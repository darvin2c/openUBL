Los siguientes ejemplos asumen que el servidor está corriendo en `http://localhost:8000`. El primer ejemplo usa `httpx`, pero puedes reemplazarlo por `requests` sin cambiar la lógica.

## Crear y firmar una factura

<CodeTabs tabs={[
  {
    id: 'python',
    label: 'Python',
    lang: 'python',
    code: `import os
import httpx

from openubl.models import Invoice, Proveedor, Cliente, DocumentoVentaDetalle

BASE_URL = "http://localhost:8000"

invoice = Invoice(
    serie="F001",
    numero=1,
    proveedor=Proveedor(
        ruc="20100111111",
        razonSocial="Mi Empresa S.A.C.",
    ),
    cliente=Cliente(
        nombre="Juan Perez",
        numeroDocumentoIdentidad="12345678",
        tipoDocumentoIdentidad="1",
    ),
    detalles=[
        DocumentoVentaDetalle(
            descripcion="Producto A",
            cantidad=2,
            precio=100.00,
        )
    ],
    moneda="PEN",
)


def main():
    with httpx.Client(base_url=BASE_URL) as client:
        # 1. Crear la factura
        r = client.post("/api/v1/invoice/create", json=invoice.model_dump(mode="json"))
        r.raise_for_status()
        xml = r.json()["xml"]
        print("XML generado:\\n", xml)

        # 2. Firmar el XML
        cert_pem = os.getenv("CERT_PEM", "")
        key_pem = os.getenv("KEY_PEM", "")

        r2 = client.post(
            "/api/v1/sign",
            json={
                "xml": xml,
                "cert_pem": cert_pem,
                "key_pem": key_pem,
                "signature_id": "SignSUNAT",
            },
        )
        r2.raise_for_status()
        signed_xml = r2.json()["signed_xml"]
        print("XML firmado:\\n", signed_xml)


if __name__ == "__main__":
    main()`,
  },
  {
    id: 'typescript',
    label: 'TypeScript',
    lang: 'typescript',
    code: `import { client, zInvoice, zSignXmlApiV1SignPostBody } from "@openubl/sdk";

const invoice = zInvoice.parse({
  serie: "F001",
  numero: 1,
  tipoOperacion: "0101",
  moneda: "PEN",
  proveedor: {
    ruc: "20100111111",
    razonSocial: "Mi Empresa S.A.C.",
  },
  cliente: {
    nombre: "Juan Perez",
    numeroDocumentoIdentidad: "12345678",
    tipoDocumentoIdentidad: "1",
  },
  detalles: [
    {
      descripcion: "Producto A",
      cantidad: 2,
      precio: 100.00,
      unidadMedida: "NIU",
      tipoAfectacionIGV: "10",
    },
  ],
});

const { data, error } = await client.post("/api/v1/invoice/create", {
  body: invoice,
});

if (error) {
  throw new Error(JSON.stringify(error));
}

const xml = data.xml;

const body = zSignXmlApiV1SignPostBody.parse({
  xml,
  cert_pem: process.env.CERT_PEM,
  key_pem: process.env.KEY_PEM,
  signature_id: "SignSUNAT",
});

const { data: signData, error: signError } = await client.post("/api/v1/sign", {
  body,
});

if (signError) {
  throw new Error(JSON.stringify(signError));
}

console.log(signData.signed_xml);`,
  },
  {
    id: 'curl',
    label: 'cURL',
    lang: 'bash',

  -d '{
    "serie": "F001",
    "numero": 1,
    "tipoOperacion": "0101",
    "moneda": "PEN",
    "proveedor": {
      "ruc": "20100111111",
      "razonSocial": "Mi Empresa S.A.C."
    },
    "cliente": {
      "nombre": "Juan Perez",
      "numeroDocumentoIdentidad": "12345678",
      "tipoDocumentoIdentidad": "1"
    },
    "detalles": [
      {
        "descripcion": "Producto A",
        "cantidad": 2,
        "precio": 100.00,
        "unidadMedida": "NIU",
        "tipoAfectacionIGV": "10"
      }
    ]
  }' | jq -r '.xml')

# 2. Firmar el XML
curl -X POST "http://localhost:8000/api/v1/sign" \\
  -H "Content-Type: application/json" \\
  -d "{
    \"xml\": \"$XML\",
    \"cert_pem\": \"$CERT_PEM\",
    \"key_pem\": \"$KEY_PEM\",
    \"signature_id\": \"SignSUNAT\"

<CodeTabs tabs={[

    id: 'python',

from openubl.models import VoidedDocuments, VoidedDocumentsItem, Proveedor
from datetime import date

BASE_URL = "http://localhost:8000"

proveedor = Proveedor(
    ruc="20100111111",
    razonSocial="Mi Empresa S.A.C.",
)

voided = VoidedDocuments(
    numero=1,
    fechaEmisionComprobantes=date(2024, 1, 15),
    proveedor=proveedor,
    comprobantes=[
        VoidedDocumentsItem(
            serie="F001",
            numero=1,
            tipoComprobante="01",
            descripcionSustento="Error en datos del cliente",
        )
    ],
)

with httpx.Client(base_url=BASE_URL) as client:
    r = client.post("/api/v1/voided-documents/create", json=voided.model_dump(mode="json"))

    print(r.json()["xml"])`,
